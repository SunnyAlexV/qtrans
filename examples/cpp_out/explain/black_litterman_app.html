82def variance_ratio(returns: Mat, h: int, shrink_c: float=10.0, lo: float=0.25, hi: float=4.0) -> Vec:
83 """Per-asset variance ratio VR(h) = Var(h-period) / (h x Var(1-period)).
84
85 The whole reason a holding period should change a portfolio. Scaling daily
86 variance by h assumes returns are independent from one day to the next. They
87 are not: an asset that mean-reverts is LESS risky over a year than its daily
88 volatility implies (VR < 1) and deserves more weight in a long hold, while a
89 trending one is MORE risky (VR > 1) and deserves less. Under strict i.i.d.
90 returns VR = 1 at every horizon and the holding period genuinely does not
91 matter -- which is why the app used to give the same answer for a week and a
92 year.
93
94 Overlapping windows mean the estimate is far noisier than the raw sample
95 size suggests: 3,780 daily observations contain only ~15 independent annual
96 ones. So VR is shrunk toward 1 by the effective sample count, which leaves a
97 weekly horizon essentially untouched and pulls a yearly one most of the way
98 back unless the evidence is strong. Clipped to [lo, hi] so a single freak
99 window cannot rescale a book.
100 """
101 R = np.asarray(returns, dtype=float)
102 h = max(1, int(h))
103 if R.ndim == 1:
104 R = R[:, None]
105 n = R.shape[1]
106 if h == 1 or len(R) <= 2 * h:
107 return np.ones(n)
108 var1 = np.var(R, axis=0, ddof=1)
109 Rh = horizon_returns(R, h)
110 if len(Rh) < 3:
111 return np.ones(n)
112 varh = np.var(Rh, axis=0, ddof=1)
113 with np.errstate(divide="ignore", invalid="ignore"):
114 vr = np.where(var1 > 1e-18, varh / (h * var1), 1.0)
115 vr = np.where(np.isfinite(vr), vr, 1.0)
116 # Effective independent observations at this horizon, not the overlapping count.
117 n_eff = len(R) / float(h)
118 k = n_eff / (n_eff + float(shrink_c))
119 vr = 1.0 + (vr - 1.0) * k

283def buy_and_hold_returns(R: Mat, w: Vec) -> tuple[Vec, Vec]:
284 """Period returns of a basket bought at weights `w` and then LEFT ALONE.
285
286 The obvious-looking `R @ w` is not this. Applying a fixed weight vector to
287 every period silently rebalances the portfolio back to `w` every single
288 period — so a 'yearly rebalanced' strategy computed that way is really a
289 daily-rebalanced one that happens to re-optimise once a year, and every
290 rebalance frequency collapses to nearly the same result.
291
292 Here the holdings drift: each asset's value compounds at its own return, and
293 the portfolio return is the change in the total.
294 """
295 R = np.asarray(R, dtype=float)
296 w = np.asarray(w, dtype=float)
297 if R.size == 0:
298 return np.zeros(0), w.copy()
299 growth = np.cumprod(1.0 + R, axis=0) # per-asset value path
300 V = growth @ w # portfolio value path, V(0-)=1
301 prev = np.concatenate(([float(np.sum(w))], V[:-1]))
302 with np.errstate(divide="ignore", invalid="ignore"):
303 r = np.where(np.abs(prev) > 1e-12, V / prev - 1.0, 0.0)
304 # weights the portfolio has DRIFTED to by the end — what the next rebalance
305 # actually trades away from, and therefore what turnover must be measured
306 # against.
307 end_val = w * growth[-1]
308 tot = float(np.sum(end_val))
309 w_end = end_val / tot if abs(tot) > 1e-12 else w.copy()
310 return r, w_end
